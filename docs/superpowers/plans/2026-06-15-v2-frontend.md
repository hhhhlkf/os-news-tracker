- [ ] **Step 2: Create `frontend/src/auth.ts`**

```typescript
import type { User } from "./types";

const TOKEN_KEY = "os_tracker_token";
const USER_KEY  = "os_tracker_user";

export function getToken(): string | null {
  return localStorage.getItem(TOKEN_KEY);
}

export function getUser(): User | null {
  const raw = localStorage.getItem(USER_KEY);
  if (!raw) return null;
  try { return JSON.parse(raw) as User; } catch { return null; }
}

export function setAuth(token: string, user: User): void {
  localStorage.setItem(TOKEN_KEY, token);
  localStorage.setItem(USER_KEY, JSON.stringify(user));
}

export function clearAuth(): void {
  localStorage.removeItem(TOKEN_KEY);
  localStorage.removeItem(USER_KEY);
}

export function authHeaders(): Record<string, string> {
  const token = getToken();
  return token ? { Authorization: `Bearer ${token}` } : {};
}
```

```tsx
import { BrowserRouter, Routes, Route, Navigate } from "react-router-dom";
import { QueryClient, QueryClientProvider } from "@tanstack/react-query";
import { HomePage } from "./pages/HomePage";
import { LoginPage } from "./pages/LoginPage";
import { RegisterPage } from "./pages/RegisterPage";
import { ProfileSettingsPage } from "./pages/ProfileSettingsPage";
import { AgentSourcesPage } from "./pages/AgentSourcesPage";
import { DigestPage } from "./pages/DigestPage";
import { getToken } from "./auth";

const queryClient = new QueryClient();

function RequireAuth({ children }: { children: React.ReactNode }) {
  if (!getToken()) return <Navigate to="/login" replace />;
  return <>{children}</>;
}

export default function App() {
  return (
    <QueryClientProvider client={queryClient}>
      <BrowserRouter>
        <Routes>
          <Route path="/login" element={<LoginPage />} />
          <Route path="/register" element={<RegisterPage />} />
          <Route path="/" element={<RequireAuth><HomePage /></RequireAuth>} />
          <Route path="/settings/profile" element={<RequireAuth><ProfileSettingsPage /></RequireAuth>} />
          <Route path="/settings/agent-sources" element={<RequireAuth><AgentSourcesPage /></RequireAuth>} />
          <Route path="/digest" element={<RequireAuth><DigestPage /></RequireAuth>} />
          <Route path="*" element={<Navigate to="/" replace />} />
        </Routes>
      </BrowserRouter>
    </QueryClientProvider>
  );
}
```

```tsx
import type { ItemSummary } from "../types";
import { ImportanceBadge } from "./ImportanceBadge";
import { InfoTypeBadge } from "./InfoTypeBadge";

function scoreColor(score: number): string {
  if (score >= 80) return "var(--signal-high)";
  if (score >= 50) return "var(--signal-mid)";
  return "var(--border)";
}

export function ItemCard({ item, onClick, sortBy }: {
  item: ItemSummary;
  onClick: () => void;
  sortBy?: "published_at" | "fetched_at";
}) {
  const isFetchedAt = sortBy === "fetched_at";
  const score = item.personalized_score;
  const borderColor = score != null ? scoreColor(score) : "var(--border)";
  const scoreTextColor = score != null ? scoreColor(score) : "var(--ink-muted)";

  return (
    <button onClick={onClick} style={{
      display: "flex", width: "100%", textAlign: "left",
      border: "1px solid var(--border-subtle)", borderLeft: `4px solid ${borderColor}`,
      borderRadius: 10, padding: "14px 14px 14px 12px",
      marginBottom: 10, background: "var(--surface)", cursor: "pointer",
      gap: 0,
    }}>
      <div style={{ flex: 1, minWidth: 0 }}>
        <div style={{ display: "flex", gap: 8, alignItems: "center", marginBottom: 6 }}>
          <ImportanceBadge value={item.importance} />
          <InfoTypeBadge value={item.info_type} />
          {item.main_category && (
            <span style={{ fontSize: 12, color: "var(--ink-muted)" }}>{item.main_category}</span>
          )}
          <span style={{ marginLeft: "auto", display: "flex", alignItems: "center", gap: 10 }}>
            {score != null && (
              <span style={{
                fontFamily: "var(--font-mono)", fontSize: 13, fontWeight: 600,
                color: scoreTextColor,
              }}>
                {score}
              </span>
            )}
            <span style={{ fontSize: 12, color: "var(--ink-muted)", fontFamily: "var(--font-mono)" }}>
              {isFetchedAt
                ? (item.fetched_at ? item.fetched_at.slice(0, 10) : "")
                : (item.published_at ? item.published_at.slice(0, 10) : "")
              }
            </span>
          </span>
        </div>
        <div style={{ fontWeight: 600, color: "var(--ink)" }}>{item.title_tldr ?? item.title}</div>
      </div>
    </button>
  );
}
```

```tsx
interface Criterion {
  id: string;
  label: string;
  enabled: boolean;
  weight: number;
  match: { keywords: string[]; sub_tags: string[]; main_category: string[]; info_type: string[]; keywords_op: string };

function CriterionCard({ c, onToggle, onDelete }: {
  c: Criterion;
  onToggle: () => void;
  onDelete: () => void;
}) {
  const tagStyle = {
    background: "var(--surface-tint)", border: "1px solid var(--border-subtle)",
    borderRadius: 4, padding: "1px 6px", fontSize: 11,
    fontFamily: "var(--font-mono)", color: "var(--ink-secondary)",
  };
  const allKeywords = [...c.match.keywords, ...c.match.sub_tags, ...c.match.main_category];

  return (
    <div style={{
      border: "1px solid var(--border)", borderLeft: `3px solid ${c.enabled ? "var(--accent)" : "var(--border)"}`,
      borderRadius: 10, padding: "14px 16px", marginBottom: 10, background: "var(--surface)",

    }}>
      <div style={{ display: "flex", alignItems: "center", gap: 12, marginBottom: 8 }}>

          {c.weight.toFixed(1)}
        </span>
        <button onClick={onToggle} style={{
          border: "1px solid var(--border)", borderRadius: 6, padding: "3px 10px",
          fontSize: 12, cursor: "pointer", background: c.enabled ? "var(--accent-subtle)" : "var(--surface-tint)",
          color: c.enabled ? "var(--accent)" : "var(--ink-muted)",
        }}>
          {c.enabled ? "启用" : "停用"}
        </button>
        <button onClick={onDelete} style={{
          border: "1px solid var(--border)", borderRadius: 6, padding: "3px 10px",
          fontSize: 12, cursor: "pointer", background: "var(--surface)", color: "var(--ink-muted)",
        }}>删除</button>

  );
}

export function ProfileSettingsPage() {
  const qc = useQueryClient();
  const { data: profile, isLoading } = useQuery({ queryKey: ["profile"], queryFn: fetchProfile });
  const [suggestions, setSuggestions] = useState<(Criterion & { reason?: string })[]>([]);
  const [suggestLoading, setSuggestLoading] = useState(false);
  const [description, setDescription] = useState("");

  const saveMutation = useMutation({
    mutationFn: (data: object) => updateProfile(data),
    onSuccess: () => void qc.invalidateQueries({ queryKey: ["profile"] }),
  });

  if (isLoading || !profile) return <div style={{ padding: 32 }}>加载中…</div>;

  const criteria: Criterion[] = profile.criteria ?? [];

  function toggleCriterion(id: string) {
    const updated = criteria.map(c => c.id === id ? { ...c, enabled: !c.enabled } : c);
    saveMutation.mutate({ ...profile, criteria: updated });
  }

  function deleteCriterion(id: string) {
    const updated = criteria.filter(c => c.id !== id);
    saveMutation.mutate({ ...profile, criteria: updated });
  }

  function acceptSuggestion(s: Criterion & { reason?: string }) {
    const { reason: _r, ...c } = s;
    const updated = [...criteria, c];
    saveMutation.mutate({ ...profile, criteria: updated });
    setSuggestions(prev => prev.filter(x => x.id !== s.id));
  }

  async function handleSuggest() {
    setSuggestLoading(true);
    try {
      const res = await suggestCriteria(description);
      setSuggestions(res.suggestions ?? []);
    } finally {
      setSuggestLoading(false);
    }
  }

  return (
    <div style={{ maxWidth: 800, margin: "0 auto", padding: 24 }}>
      <h2 style={{ color: "var(--ink)", marginBottom: 24 }}>偏好设置</h2>

          </div>
        )}
        {criteria.map(c => (
          <CriterionCard key={c.id} c={c}
            onToggle={() => toggleCriterion(c.id)}
            onDelete={() => deleteCriterion(c.id)} />
        ))}
      </section>

      {/* LLM Scoring */}
      <section style={{
        marginBottom: 32, padding: 20, background: "var(--surface)",

          <button
            onClick={() => saveMutation.mutate({ ...profile, enable_llm_scoring: !profile.enable_llm_scoring })}
            style={{
              width: 44, height: 24, borderRadius: 12, border: "none", cursor: "pointer",
```

```tsx
      ))}

      {(sources ?? []).length === 0 && !showForm && (
        <div style={{ color: "var(--ink-muted)", textAlign: "center", padding: "40px 0" }}>
          还没有智能采集源。点击「+ 添加来源」开始。
        </div>
      )}
    </div>
  );
}
```
